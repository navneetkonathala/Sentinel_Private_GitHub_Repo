from collections import deque
from statistics import median

class Sentinel:
    """Online robust detector. Feed one numeric sample at a time."""
    def __init__(self, window=51, threshold=3.5, min_samples=12):
        self.values = deque(maxlen=window)
        self.threshold = threshold
        self.min_samples = min_samples

    def update(self, value):
        if len(self.values) < self.min_samples:
            self.values.append(value)
            return {"status": "warming_up", "value": value,
                    "samples": len(self.values), "needed": self.min_samples,
                    "explanation": "Collecting normal readings; not scoring yet."}

        center = median(self.values)
        deviations = [abs(item - center) for item in self.values]
        mad = median(deviations)
        spread = max(1.4826 * mad, 1e-9)
        score = (value - center) / spread
        upper = center + self.threshold * spread
        lower = center - self.threshold * spread
        flagged = abs(score) > self.threshold
        direction = "high" if value > upper else "low" if value < lower else "within_range"

        # Never teach the baseline from a flagged point.
        if not flagged:
            self.values.append(value)

        return {
            "status": "anomaly" if flagged else "normal",
            "value": value,
            "direction": direction,
            "deviation": round(value - center, 3),
            "score": round(score, 3),
            "baseline": center,
            "upper": upper,
            "lower": lower,
            "explanation": "Compare value with the upper/lower limit to explain the alert.",
        }
